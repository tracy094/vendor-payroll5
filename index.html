Google Apps Script (GAS) 回傳腳本說明

請在您的 Google 試算表內點選 擴充功能 > Apps Script，貼上以下程式碼並部署為「Web 應用程式」（存取權限設為「所有人」）： 

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var dateStr = Utilities.formatDate(new Date(), "GMT+8", "yyyy-MM-dd");
    var sheetName = dateStr + "_請款試算";
    
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(["姓名", "廠商", "工時", "工資", "加項", "未稅小計", "營業稅5%", "含稅總額"]);
    }
    
    data.records.forEach(function(r) {
      sheet.appendRow([r.worker, r.vendor, r.hours, r.wage, r.allowances, r.subtotal, r.tax, r.total]);
    });
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
